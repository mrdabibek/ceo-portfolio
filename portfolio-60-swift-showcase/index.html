import SwiftUI, Combine
// TCA-style feature: state · action · reducer
@Observable final class TransferModel {
  var amount: Decimal = 0
  var state: SendState = .idle
  func send() async {
    state = .sending
    do { try await api.transfer(amount)
      state = .done } catch { state = .error(error) }
  }
}
struct SendView: View {
  @State var model = TransferModel()
  var body: some View {
    VStack { AmountField($model.amount)
      PayButton(state: model.state) {
        Task { await model.send() } } }
      .task { await model.loadLimits() }
  }
}
